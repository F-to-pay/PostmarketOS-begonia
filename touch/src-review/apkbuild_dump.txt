pkgrel=1
pkgdesc="Mainline kernel fork for MT6785"
arch="aarch64"
_carch="arm64"
url="https://gitlab.postmarketos.org/mt6785-mainline/linux"
license="GPL-2.0-only"
options="
	!check
	!strip
	!tracedeps
	pmb:cross-native
	pmb:kconfigcheck-community
"
makedepends="
	bison
	clang
	elfutils-dev
	findutils
	flex
	linux-headers
	lld
	llvm
	openssl-dev
	pahole
	perl
	postmarketos-installkernel
	python3
	zstd
"

# Source
_tag="$pkgver-r0"
_config="config-$_flavor.$arch"
source="
	https://gitlab.postmarketos.org/mt6785-mainline/linux/-/archive/$_tag/linux-$_tag.tar.gz
	$_config
"
builddir="$srcdir/linux-$_tag"

prepare() {
	default_prepare
	cp "$srcdir/config-$_flavor.$arch" .config
}

build() {
	unset LDFLAGS
	make ARCH="$_carch" LLVM=1 HOSTLDFLAGS="-fuse-ld=lld" \
		KBUILD_BUILD_VERSION="$((pkgrel + 1))-$_flavor"
}

package() {
	mkdir -p "$pkgdir"/boot

	make dtbs_install modules_install zinstall \
		ARCH="$_carch" \
		LLVM=1 \
		INSTALL_PATH="$pkgdir"/boot/ \
		INSTALL_MOD_PATH="$pkgdir"/usr \
		INSTALL_MOD_STRIP=1 \
		INSTALL_DTBS_PATH="$pkgdir"/boot/dtbs
	rm -f "$pkgdir"/usr/lib/modules/*/build "$pkgdir"/usr/lib/modules/*/source

	install -D "$builddir"/include/config/kernel.release \
		"$pkgdir"/usr/share/kernel/"$_flavor"/kernel.release
}

================ testing/device-xiaomi-begonia/APKBUILD
# Reference: <https://postmarketos.org/devicepkg>
maintainer="Dinolek <git@dinolek.me>"
pkgname=device-xiaomi-begonia
pkgdesc="Xiaomi Redmi Note 8 Pro"
pkgver=6
pkgrel=0
url="https://postmarketos.org"
license="MIT"
arch="aarch64"
options="!check !archcheck"
depends="
	firmware-xiaomi-begonia-touchscreen
	linux-postmarketos-mediatek-mt6785
	mesa-vulkan-panfrost
	mkbootimg
	postmarketos-base
"
makedepends="android-tools devicepkg-dev dtc"
source="deviceinfo kernel-cmdline.conf modules-initfs"

build() {
	devicepkg_build $startdir $pkgname

	# begonia bootloader needs empty dtbo, build it and add as recovery_dtbo to final image
	mkdir "$builddir"
	echo '/dts-v1/;/{};' | dtc -I dts -O dtb -o "$builddir/empty.dtb"
	mkdtboimg create "$builddir/empty.dtbo" "$builddir/empty.dtb"
}

package() {
	devicepkg_package $startdir $pkgname

	mkdir -p "$pkgdir/boot"
	install -Dm644 "$builddir/empty.dtbo" \
		"$pkgdir/boot/"
}
